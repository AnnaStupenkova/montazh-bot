#cloud-config
# Установка ДВУХ ботов на один сервер: «Дневник Состояния» и «Монтаж».
# Замени значения в кавычках там, где написано ВСТАВЬ / ПРИДУМАЙ. Больше ничего не меняй.
write_files:
  - path: /root/boty.env
    permissions: "0600"
    content: |
      # --- Дневник Состояния ---
      DNEVNIK_TOKEN="ВСТАВЬ_КЛЮЧ_БОТА_ДНЕВНИКА"
      DNEVNIK_CODE="ПРИДУМАЙ_КОДОВОЕ_СЛОВО_ДНЕВНИКА"
      PROGRAM_START="2026-10-15"
      DNEVNIK_REPO="https://github.com/AnnaStupenkova/dnevnik-bot.git"
      # --- Монтаж ---
      MONTAZH_TOKEN="ВСТАВЬ_КЛЮЧ_БОТА_МОНТАЖА"
      MONTAZH_CODE="ПРИДУМАЙ_КОДОВОЕ_СЛОВО_МОНТАЖА"
      PIXABAY_KEY="58508510-c5c9ba8f9c6a67c66b1883abb"
      GIGACHAT_KEY=""
      MONTAZH_REPO="https://github.com/AnnaStupenkova/montazh-bot.git"
  - path: /etc/systemd/system/dnevnik.service
    content: |
      [Unit]
      Description=Dnevnik Sostoyaniya bot
      After=network-online.target
      Wants=network-online.target

      [Service]
      WorkingDirectory=/opt/dnevnik
      ExecStart=/bin/bash -c 'cd /opt/dnevnik && exec venv/bin/python "$(ls bot*.py | head -1)"'
      Restart=always
      RestartSec=5
      Environment=PYTHONUNBUFFERED=1

      [Install]
      WantedBy=multi-user.target
  - path: /etc/systemd/system/tgapi.service
    content: |
      [Unit]
      Description=Local Telegram Bot API (big files)
      After=network-online.target
      Wants=network-online.target

      [Service]
      EnvironmentFile=/root/tgapi.env
      ExecStart=/usr/local/bin/telegram-bot-api --api-id=${TG_API_ID} --api-hash=${TG_API_HASH} --local --http-port=8081 --http-ip-address=127.0.0.1 --dir=/var/lib/telegram-bot-api --temp-dir=/var/tmp/tgapi
      Restart=always
      RestartSec=5

      [Install]
      WantedBy=multi-user.target
  - path: /etc/systemd/system/montazh.service
    content: |
      [Unit]
      Description=Montazh reels bot
      After=network-online.target tgapi.service
      Wants=network-online.target

      [Service]
      WorkingDirectory=/opt/montazh
      EnvironmentFile=/opt/montazh/.env
      ExecStart=/opt/montazh/venv/bin/python bot.py
      Restart=always
      RestartSec=5
      Nice=10
      CPUWeight=50
      Environment=PYTHONUNBUFFERED=1

      [Install]
      WantedBy=multi-user.target
  - path: /root/install.sh
    permissions: "0700"
    content: |
      #!/bin/bash
      exec > /root/install.log 2>&1
      set -x
      sed -i 's/\r$//' /root/boty.env
      . /root/boty.env
      export DEBIAN_FRONTEND=noninteractive
      # запас памяти на время установки
      if [ ! -f /swapfile ]; then fallocate -l 4G /swapfile && chmod 600 /swapfile && mkswap /swapfile && swapon /swapfile && echo '/swapfile none swap sw 0 0' >> /etc/fstab; fi
      apt-get update
      apt-get install -y python3-venv python3-dev git ffmpeg curl ca-certificates bzip2 unzip \
          make cmake g++ gperf zlib1g-dev libssl-dev
      # сертификат Минцифры (нужен для GigaChat)
      curl -fsSL https://gu-st.ru/content/lending/russian_trusted_root_ca_pem.crt -o /usr/local/share/ca-certificates/russian_trusted_root_ca.crt && update-ca-certificates || true

      # ===== 1. Дневник =====
      rm -rf /opt/dnevnik
      git clone "$DNEVNIK_REPO" /opt/dnevnik
      cd /opt/dnevnik
      python3 -m venv venv
      venv/bin/pip install -r "$(ls requirements*.txt | head -1)"
      printf 'BOT_TOKEN=%s\nADMIN_CODE=%s\nPROGRAM_START=%s\n' "$DNEVNIK_TOKEN" "$DNEVNIK_CODE" "$PROGRAM_START" > .env
      chmod 600 .env
      systemctl daemon-reload
      systemctl enable --now dnevnik

      # ===== 2. Монтаж: код и модели =====
      rm -rf /opt/montazh
      git clone "$MONTAZH_REPO" /opt/montazh
      cd /opt/montazh
      # если код загружен архивом — распаковать
      Z=$(ls *.zip 2>/dev/null | head -1); [ -n "$Z" ] && unzip -o "$Z"
      python3 -m venv venv
      venv/bin/pip install -r requirements.txt
      mkdir -p /opt/montazh-models && cd /opt/montazh-models
      R=https://github.com/k2-fsa/sherpa-onnx/releases/download/asr-models
      curl -fL -o silero_vad.onnx $R/silero_vad.onnx
      M=sherpa-onnx-nemo-transducer-giga-am-v2-russian-2025-04-19
      for i in 1 2 3; do curl -fL -o $M.tar.bz2 $R/$M.tar.bz2 && tar xjf $M.tar.bz2 && rm $M.tar.bz2 && break; sleep 10; done
      cat > /opt/montazh/.env <<EOF
      BOT_TOKEN=$MONTAZH_TOKEN
      ADMIN_CODE=$MONTAZH_CODE
      DATA=/opt/montazh-data
      MODELS=/opt/montazh-models
      PIXABAY_KEY=$PIXABAY_KEY
      GIGACHAT_KEY=$GIGACHAT_KEY
      EOF
      chmod 600 /opt/montazh/.env

      # бот монтажа запускается сразу (файлы до 20 МБ)
      systemctl daemon-reload
      systemctl enable --now montazh

      # ===== 3. Локальный сервер Telegram (файлы больше 20 МБ) =====
      # собирается в фоне ~40-60 мин; включается потом командой боту: /bigfiles api_id api_hash
      mkdir -p /var/lib/telegram-bot-api /var/tmp/tgapi
      cd /root && rm -rf telegram-bot-api
      git clone --recursive https://github.com/tdlib/telegram-bot-api.git
      cd telegram-bot-api && mkdir build && cd build
      cmake -DCMAKE_BUILD_TYPE=Release -DCMAKE_INSTALL_PREFIX:PATH=/usr/local ..
      cmake --build . --target install -j 2
      echo "ГОТОВО"
runcmd:
  - [bash, /root/install.sh]
